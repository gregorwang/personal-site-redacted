## 2. 技术 SEO 优化

### 2.1 多语言支持（hreflang）

**问题**: 网站有中英文内容，但缺少 hreflang 标签。

**解决方案**:
在 `app/utils/seo.ts` 的 `generateMeta` 函数中添加：

```typescript
export interface SEOConfig {
  // ... 现有字段
  alternateLanguages?: Array<{ lang: string; url: string }>;
}

export function generateMeta(config: SEOConfig): MetaDescriptor[] {
  const meta: MetaDescriptor[] = [
    // ... 现有 meta 标签
  ];

  // 添加 hreflang 标签
  if (config.alternateLanguages) {
    config.alternateLanguages.forEach(({ lang, url }) => {
      meta.push({
        tagName: "link",
        rel: "alternate",
        hreflang: lang,
        href: url,
      });
    });
  }

  return meta;
}

// 使用示例
export const pageMeta = {
  home: () =>
    generateMeta({
      title: "汪家俊的个人网站 - 将我的爱弃置于此",
      description: "...",
      url: "https://wangjiajun.asia",
      alternateLanguages: [
        { lang: "zh-CN", url: "https://wangjiajun.asia" },
        { lang: "en", url: "https://wangjiajun.asia/en" },
        { lang: "x-default", url: "https://wangjiajun.asia" },
      ],
    }),
};
```

### 2.2 面包屑导航和结构化数据

**建议**: 添加面包屑导航提升用户体验和 SEO。

**实现**:
```typescript
// app/components/Breadcrumbs.tsx
export function Breadcrumbs({ items }: { items: Array<{ name: string; url: string }> }) {
  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "BreadcrumbList",
    "itemListElement": items.map((item, index) => ({
      "@type": "ListItem",
      "position": index + 1,
      "name": item.name,
      "item": item.url,
    })),
  };

  return (
    <>
      <script
        type="application/ld+json"
        dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
      />
      <nav aria-label="Breadcrumb">
        {items.map((item, index) => (
          <span key={index}>
            {index > 0 && " / "}
            <Link to={item.url}>{item.name}</Link>
          </span>
        ))}
      </nav>
    </>
  );
}
```
